<div class="container-xxl">
    <div class="row postcontent">
        <div class="col-12">
            <h1 class="mb-5">404 - Page or File Not Found</h1>
            <p>
                Oh no,  this is embarrassing 🙈! The page you're looking for doesn't exist or has been moved to another address.
            </p>
            <div id="nearest-pages" hidden>
                <p>You might find what you're looking for here:</p>
                <ul id="nearest-pages-list"></ul>
            </div>
            <p>If you think the address <code id="invalid-address"></code> should be valid, <a href="https://www.stride3d.net/contact/">let us know</a>.</p>
    <script>
        document.getElementById("invalid-address").textContent = window.location.href;
        // This page is served for any missing url, whatever its version, language and section.
        // Before loading docfx (see PostProcessing-Fixing404AbsolutePath in BuildDocs.ps1), it points the navbar
        // and the sidebar to the tables of contents of the requested version and section, when they exist.
        (async () => {
            const getMeta = name => document.querySelector(`meta[name="${name}"]`);
            // Returns the url (after redirects) if it exists
            const exists = async url => {
                try {
                    const response = await fetch(url, { method: 'HEAD' });
                    return response.ok ? response.url : null;
                } catch {
                    return null;
                }
            };
            const docfxUrl = getMeta('stride:docfx')?.content;
            if (!docfxUrl) return;
            // Base url of the docs serving this page, i.e. /4.4/en/
            const fallbackBase = new URL('../', new URL(docfxUrl, location.href)).pathname;
            // i.e. /4.3/en/contributors/engine/missing.html
            const match = location.pathname.match(/^\/(latest|\d+\.\d+)\/([a-z]{2})\/(.*)$/);
            const base = match ? `/${match[1]}/${match[2]}/` : fallbackBase;
            const pathSegments = match ? match[3].split('/') : [];
            const section = pathSegments.length > 1 ? pathSegments[0] : null;
            // Documentation of 4.0 and older has no toc.json
            const navbarBase = await exists(`${base}toc.json`) ? base : fallbackBase;
            const sectionBase = section && await exists(`${navbarBase}${section}/toc.json`) ? `${navbarBase}${section}/` : navbarBase;
            getMeta('docfx:rel').content = navbarBase;
            getMeta('docfx:navrel').content = `${navbarBase}toc.html`;
            getMeta('docfx:tocrel').content = `${sectionBase}toc.html`;
            await import(docfxUrl);
            // Suggest the closest existing parent page, and the same page in the latest version
            const parentPages = [];
            for (let i = pathSegments.length - 1; i > 0; i--) {
                parentPages.push(`${base}${pathSegments.slice(0, i).join('/')}/index.html`);
            }
            const suggestions = [(await Promise.all(parentPages.map(exists))).find(url => url)];
            if (match && match[1] !== 'latest') {
                suggestions.push(await exists(`/latest/${match[2]}/${match[3]}`));
            }
            const list = document.getElementById('nearest-pages-list');
            for (const url of new Set(suggestions.filter(url => url))) {
                const html = await (await fetch(url)).text();
                const link = document.createElement('a');
                link.href = url;
                link.textContent = new DOMParser().parseFromString(html, 'text/html').title || url;
                const item = document.createElement('li');
                item.appendChild(link);
                list.appendChild(item);
            }
            document.getElementById('nearest-pages').hidden = list.children.length === 0;
        })();
    </script>
        </div>
    </div>
</div>
